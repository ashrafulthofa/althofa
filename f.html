int n = 5;

for(int i=0;i<n-1;i++) {
    for(int j=0;j<n-i-1;j++) {
        if(arr[j] > arr[j+1]) {
            swap(arr[j], arr[j+1]);

Insertion Sort

```

using namespace std;

int main() {
int arr[] = {5,3,8,4,2};
int n=5;

```
for(int i=1;i<n;i++) {
    int key=arr[i];
    int j=i-1;

    while(j>=0 && arr[j]>key) {
        arr[j+1]=arr[j];
        j--;
    }

    arr[j+1]=key;
}

cout<<"Sorted Array: ";
for(int i=0;i<n;i++)
    cout<<arr[i]<<" ";
```

}
